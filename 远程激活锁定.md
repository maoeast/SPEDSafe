# 远程激活锁定功能设计文档

本文档记录特教安全教育资源仪的远程激活锁定功能设计方案，包括现有的激活码存储机制和三种远程锁定的实现方案。

## ?? 当前激活码存储机制

### 存储位置

激活码信息**保存在 Electron-Store**（加密的 JSON 文件），而**不是注册表**：

```javascript
const licenseStore = new Store({
    name: 'license-data',
    encryptionKey: 'sped-safe-encryption-key-2025'
});
```

**实际存储路径**：
- **Windows**：`C:\Users\{用户名}\AppData\Roaming\spedsafe\license-data.json`（加密）
- **macOS**：`~/Library/Application Support/spedsafe/license-data.json`（加密）
- **Linux**：`~/.config/spedsafe/license-data.json`（加密）

### 存储的数据结构

```json
{
    "activation": {
        "data": "加密的AES数据",
        "signature": "HMAC-SHA256防篡改签名"
    },
    "firstRun": 1732594800000
}
```

**解密后的激活数据**：

```javascript
{
    type: 'full',                          // 激活类型：trial, full
    version: '1.0',                        // 版本
    machineId: 'ABC123DEF456',            // 绑定的机器码
    createdAt: 1732594800000,             // 激活创建时间
    expireAt: 1764130800000,              // 激活过期时间
    permanent: false,                      // 是否永久激活
    activatedAt: 1732594800000,           // 本地激活时间
    currentMachineId: 'ABC123DEF456',     // 当前机器码
    lastVerifiedAt: 1732594800000         // 最后验证时间
}
```

### 现有安全机制

? **AES加密**：激活数据使用 AES 加密  
? **HMAC防篡改**：添加 HMAC-SHA256 签名防止本地篡改  
? **RSA验证**：激活码本身使用 RSA-2048 签名验证  
? **机器绑定**：激活码与机器硬件信息绑定

## ?? 远程锁定方案设计

### 方案 A：云服务式远程锁定（需要网络连接）

#### 特点
- 实时锁定：激活码可立即被撤销
- 需要网络连接
- 中心化管理，便于跟踪

#### 实现代码

```javascript
// 改进的激活验证，支持远程检查
async function verifyLicenseWithRemoteCheck(licenseKey) {
    // 1. 本地离线验证（现有逻辑）
    const licenseData = decryptLicenseKey(licenseKey);
    if (!licenseData) {
        return { success: false, message: '激活码无效或已损坏' };
    }

    const currentMachineId = getMachineFingerprint();

    // 2. 验证机器码匹配（现有逻辑）
    if (!verifyMachineMatch(licenseData.machineId, currentMachineId)) {
        return {
            success: false,
            message: '激活码与当前机器不匹配',
            details: `激活码绑定机器: ${licenseData.machineId}\n当前机器码: ${currentMachineId}`
        };
    }

    // 3. 验证是否过期（现有逻辑）
    if (licenseData.expireAt && Date.now() > licenseData.expireAt) {
        return {
            success: false,
            message: '激活码已过期',
            details: `过期时间: ${new Date(licenseData.expireAt).toLocaleString('zh-CN')}`
        };
    }

    // 4. 如果本地验证通过，尝试远程检查
    if (hasNetworkConnection()) {
        try {
            const remoteStatus = await checkRemoteRevocation(licenseData.machineId);
            if (remoteStatus.revoked) {
                return {
                    success: false,
                    message: '该激活码已被远程撤销',
                    reason: remoteStatus.reason,
                    revokedAt: remoteStatus.revokedAt,
                    remote: true
                };
            }
        } catch (error) {
            // 网络错误时，继续允许使用（离线优先策略）
            console.warn('远程检查失败，继续使用本地激活信息:', error);
        }
    }

    // 5. 保存激活信息
    const activationData = {
        ...licenseData,
        activatedAt: Date.now(),
        currentMachineId: currentMachineId,
        lastVerifiedAt: Date.now()
    };

    const encrypted = encryptActivationData(activationData);
    licenseStore.set('activation', encrypted);

    return {
        success: true,
        message: '激活成功！',
        data: {
            type: licenseData.type,
            isPermanent: licenseData.permanent || false,
            expireAt: licenseData.expireAt,
            features: licenseData.features
        }
    };
}

// 检查网络连接
function hasNetworkConnection() {
    return require('os').networkInterfaces();
}

// 向服务器检查撤销状态
async function checkRemoteRevocation(machineId) {
    const timeout = 5000; // 5秒超时
    
    try {
        const controller = new AbortController();
        const id = setTimeout(() => controller.abort(), timeout);

        const response = await fetch('https://api.spedsafe.com/check-revocation', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                machineId,
                timestamp: Date.now()
            }),
            signal: controller.signal
        });

        clearTimeout(id);

        if (!response.ok) {
            throw new Error(`服务器返回 ${response.status}`);
        }

        return await response.json();
    } catch (error) {
        console.error('远程撤销检查失败:', error);
        throw error;
    }
}

// 远程撤销激活码
async function revokeRemoteLicense(machineId, reason = '激活码已撤销') {
    try {
        const response = await fetch('https://api.spedsafe.com/revoke', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': 'Bearer ADMIN_TOKEN'
            },
            body: JSON.stringify({
                machineId,
                revokedAt: new Date().toISOString(),
                reason,
                revokedBy: 'admin'
            })
        });

        if (!response.ok) {
            throw new Error(`撤销失败: ${response.status}`);
        }

        return {
            success: true,
            message: '激活码已远程撤销',
            data: await response.json()
        };
    } catch (error) {
        console.error('远程撤销失败:', error);
        return {
            success: false,
            message: '远程撤销失败',
            error: error.message
        };
    }
}

// IPC处理：远程撤销请求
ipcMain.handle('revoke-license-remote', async (event, machineId, reason) => {
    return await revokeRemoteLicense(machineId, reason);
});

// IPC处理：检查激活状态（包含远程检查）
ipcMain.handle('verify-license-with-remote', async (event, licenseKey) => {
    return await verifyLicenseWithRemoteCheck(licenseKey);
});
```

### 方案 B：本地撤销列表（完全离线）

#### 特点
- 完全离线，不依赖网络
- 撤销列表可通过 U 盘或其他方式更新
- 适合教育机构内网环境

#### 实现代码

```javascript
// 初始化撤销列表存储
const revocationListStore = new Store({
    name: 'revocation-list',
    defaults: {
        revokedMachines: [],           // 已撤销的机器ID列表
        revokedMachinesDetail: {},     // 详细撤销信息
        lastUpdated: null,              // 最后更新时间
        version: 1                      // 列表版本
    }
});

// 检查机器是否在撤销列表中
function isInRevocationList(machineId) {
    const list = revocationListStore.get('revokedMachines', []);
    return list.includes(machineId.toUpperCase());
}

// 获取撤销详情
function getRevocationDetails(machineId) {
    const details = revocationListStore.get('revokedMachinesDetail', {});
    return details[machineId.toUpperCase()] || null;
}

// 本地撤销某台机器
function revokeMachine(machineId, reason = '手动撤销') {
    const machineIdUpper = machineId.toUpperCase();
    const list = revocationListStore.get('revokedMachines', []);
    const details = revocationListStore.get('revokedMachinesDetail', {});

    if (!list.includes(machineIdUpper)) {
        list.push(machineIdUpper);
        details[machineIdUpper] = {
            machineId: machineIdUpper,
            revokedAt: new Date().toISOString(),
            reason: reason,
            revokedBy: 'local'
        };

        revocationListStore.set('revokedMachines', list);
        revocationListStore.set('revokedMachinesDetail', details);

        console.log(`? 已撤销机器: ${machineIdUpper}`);
        return true;
    }
    return false;
}

// 取消撤销
function unrevokeMachine(machineId) {
    const machineIdUpper = machineId.toUpperCase();
    const list = revocationListStore.get('revokedMachines', []);
    const details = revocationListStore.get('revokedMachinesDetail', {});

    const index = list.indexOf(machineIdUpper);
    if (index > -1) {
        list.splice(index, 1);
        delete details[machineIdUpper];

        revocationListStore.set('revokedMachines', list);
        revocationListStore.set('revokedMachinesDetail', details);

        console.log(`? 已取消撤销: ${machineIdUpper}`);
        return true;
    }
    return false;
}

// 获取所有撤销列表
function getAllRevocations() {
    return {
        revokedMachines: revocationListStore.get('revokedMachines', []),
        details: revocationListStore.get('revokedMachinesDetail', {}),
        lastUpdated: revocationListStore.get('lastUpdated'),
        version: revocationListStore.get('version', 1)
    };
}

// 导入撤销列表（从文件或其他来源）
function importRevocationList(listData) {
    try {
        if (!listData.revokedMachines || !Array.isArray(listData.revokedMachines)) {
            throw new Error('撤销列表格式无效');
        }

        revocationListStore.set('revokedMachines', listData.revokedMachines);
        revocationListStore.set('revokedMachinesDetail', listData.revokedMachinesDetail || {});
        revocationListStore.set('lastUpdated', new Date().toISOString());
        revocationListStore.set('version', listData.version || 1);

        console.log(`? 已导入撤销列表，共 ${listData.revokedMachines.length} 台机器`);
        return { success: true, count: listData.revokedMachines.length };
    } catch (error) {
        console.error('导入撤销列表失败:', error);
        return { success: false, error: error.message };
    }
}

// 导出撤销列表（用于备份或传播）
function exportRevocationList() {
    return {
        revokedMachines: revocationListStore.get('revokedMachines', []),
        revokedMachinesDetail: revocationListStore.get('revokedMachinesDetail', {}),
        lastUpdated: revocationListStore.get('lastUpdated'),
        version: revocationListStore.get('version', 1),
        exportedAt: new Date().toISOString()
    };
}

// 改进的激活验证（包含本地撤销列表检查）
function verifyLicenseKeyWithRevocation(licenseKey) {
    // 1. 解密激活码
    const licenseData = decryptLicenseKey(licenseKey);
    if (!licenseData) {
        return { success: false, message: '激活码无效或已损坏' };
    }

    // 2. 检查本地撤销列表
    if (isInRevocationList(licenseData.machineId)) {
        const revocationInfo = getRevocationDetails(licenseData.machineId);
        return {
            success: false,
            message: '该激活码已被撤销',
            revoked: true,
            revocationInfo: revocationInfo
        };
    }

    // 3. 获取当前机器码
    const currentMachineId = getMachineFingerprint();

    // 4. 验证机器码匹配
    if (!verifyMachineMatch(licenseData.machineId, currentMachineId)) {
        return {
            success: false,
            message: '激活码与当前机器不匹配',
            details: `激活码绑定机器: ${licenseData.machineId}\n当前机器码: ${currentMachineId}`
        };
    }

    // 5. 验证是否过期
    if (licenseData.expireAt && Date.now() > licenseData.expireAt) {
        return {
            success: false,
            message: '激活码已过期',
            details: `过期时间: ${new Date(licenseData.expireAt).toLocaleString('zh-CN')}`
        };
    }

    // 6. 保存激活信息
    const activationData = {
        ...licenseData,
        activatedAt: Date.now(),
        currentMachineId: currentMachineId,
        lastVerifiedAt: Date.now()
    };

    const encrypted = encryptActivationData(activationData);
    licenseStore.set('activation', encrypted);

    return {
        success: true,
        message: '激活成功！',
        data: {
            type: licenseData.type,
            isPermanent: licenseData.permanent || false,
            expireAt: licenseData.expireAt,
            features: licenseData.features
        }
    };
}
```

### 方案 C：混合方案（推荐）

#### 特点
- 结合本地离线验证 + 可选的网络撤销检查
- 离线优先，网络可选
- 既保证离线使用，又支持远程控制

#### 实现代码

```javascript
// 混合方案的完整实现
const revocationListStore = new Store({
    name: 'revocation-list',
    defaults: {
        revokedMachines: [],
        revokedMachinesDetail: {},
        lastUpdated: null,
        lastRemoteCheck: null,
        remoteCheckInterval: 24 * 60 * 60 * 1000
    }
});

// 混合验证：本地离线 + 可选远程检查
async function verifyLicenseKeyHybrid(licenseKey) {
    // 1. 解密激活码
    const licenseData = decryptLicenseKey(licenseKey);
    if (!licenseData) {
        return { success: false, message: '激活码无效或已损坏' };
    }

    // 2. 检查本地撤销列表
    if (isInRevocationList(licenseData.machineId)) {
        const revocationInfo = getRevocationDetails(licenseData.machineId);
        return {
            success: false,
            message: '该激活码已被撤销',
            revoked: true,
            revokedLocally: true,
            revocationInfo: revocationInfo
        };
    }

    // 3. 获取当前机器码
    const currentMachineId = getMachineFingerprint();

    // 4. 验证机器码匹配
    if (!verifyMachineMatch(licenseData.machineId, currentMachineId)) {
        return {
            success: false,
            message: '激活码与当前机器不匹配',
            details: `激活码绑定机器: ${licenseData.machineId}\n当前机器码: ${currentMachineId}`
        };
    }

    // 5. 验证是否过期
    if (licenseData.expireAt && Date.now() > licenseData.expireAt) {
        return {
            success: false,
            message: '激活码已过期',
            details: `过期时间: ${new Date(licenseData.expireAt).toLocaleString('zh-CN')}`
        };
    }

    // 6. 后台进行远程检查（不阻塞）
    if (isNetworkAvailable()) {
        performBackgroundRemoteCheck(licenseData.machineId);
    }

    // 7. 保存激活信息
    const activationData = {
        ...licenseData,
        activatedAt: Date.now(),
        currentMachineId: currentMachineId,
        lastVerifiedAt: Date.now(),
        verificationMethod: 'hybrid'
    };

    const encrypted = encryptActivationData(activationData);
    licenseStore.set('activation', encrypted);

    return {
        success: true,
        message: '激活成功！',
        data: {
            type: licenseData.type,
            isPermanent: licenseData.permanent || false,
            expireAt: licenseData.expireAt,
            features: licenseData.features
        }
    };
}

// 后台执行远程检查
function performBackgroundRemoteCheck(machineId) {
    const lastCheck = revocationListStore.get('lastRemoteCheck');
    const now = Date.now();
    const interval = revocationListStore.get('remoteCheckInterval', 24 * 60 * 60 * 1000);

    if (lastCheck && (now - lastCheck) < interval) {
        return;
    }

    setImmediate(async () => {
        try {
            console.log('?? 后台同步远程撤销列表...');
            const remoteList = await fetchRemoteRevocationList();

            if (remoteList && remoteList.revokedMachines) {
                const localList = revocationListStore.get('revokedMachines', []);
                const mergedList = Array.from(new Set([
                    ...localList,
                    ...remoteList.revokedMachines
                ]));

                revocationListStore.set('revokedMachines', mergedList);
                revocationListStore.set('revokedMachinesDetail', {
                    ...revocationListStore.get('revokedMachinesDetail', {}),
                    ...remoteList.revokedMachinesDetail
                });
                revocationListStore.set('lastRemoteCheck', now);

                console.log(`? 已同步撤销列表，共 ${mergedList.length} 台机器`);
            }
        } catch (error) {
            console.warn('远程撤销列表同步失败（非致命）:', error.message);
        }
    });
}

// 获取远程撤销列表
async function fetchRemoteRevocationList() {
    const timeout = 5000;

    try {
        const controller = new AbortController();
        const id = setTimeout(() => controller.abort(), timeout);

        const response = await fetch('https://api.spedsafe.com/revocation-list', {
            method: 'GET',
            headers: { 'Content-Type': 'application/json' },
            signal: controller.signal
        });

        clearTimeout(id);

        if (!response.ok) {
            throw new Error(`服务器返回 ${response.status}`);
        }

        const data = await response.json();
        return data;
    } catch (error) {
        console.error('获取远程撤销列表失败:', error);
        return null;
    }
}

// 检查网络连接
function isNetworkAvailable() {
    try {
        const { networkInterfaces } = require('os');
        const interfaces = networkInterfaces();

        for (const name of Object.keys(interfaces)) {
            const iface = interfaces[name];
            for (const addr of iface) {
                if (addr.family === 'IPv4' && !addr.internal) {
                    return true;
                }
            }
        }
        return false;
    } catch (error) {
        return false;
    }
}

// IPC处理：混合验证
ipcMain.handle('verify-license-hybrid', async (event, licenseKey) => {
    return await verifyLicenseKeyHybrid(licenseKey);

// IPC处理：手动同步远程列表
ipcMain.handle('sync-revocation-list', async (event) => {
    try {
        const remoteList = await fetchRemoteRevocationList();
        if (remoteList) {
            const result = importRevocationList(remoteList);
            return {
                success: true,
                message: '已同步远程撤销列表',
                ...result
            };
        }
        return {
            success: false,
            message: '获取远程列表失败'
        };
    } catch (error) {
        return {
            success: false,
            message: '同步失败',
            error: error.message
        };
    }
});
```
